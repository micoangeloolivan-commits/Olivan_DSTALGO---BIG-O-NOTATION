        private void button1_Click(object sender, EventArgs e)
        {
            int d = 100;
            int damage = 1;

            for  (int i = 0; i < d; i++)
            {
                damage += 5;
            }

            lblTotal.Text = "TOTAL DAMAGE: " + damage;
        }
